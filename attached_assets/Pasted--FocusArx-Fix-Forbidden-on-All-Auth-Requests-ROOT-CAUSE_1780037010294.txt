# FocusArx — Fix "Forbidden" on All Auth Requests

## ROOT CAUSE (confirmed by reading the code)

There are **two causes** of the 403 Forbidden error on login/register/admin:

---

### CAUSE 1 — CORS blocks all requests in production

**File:** `artifacts/api-server/src/app.ts`

In production (`NODE_ENV=production`), the CORS middleware only allows requests from `APP_URL`:

```typescript
// Current broken logic:
if (isDev) { cb(null, true); return; }  // dev: allow all
if (ALLOWED_ORIGINS.some((o) => origin === o || origin.endsWith(".replit.dev"))) {
  cb(null, true);  // prod: only APP_URL
} else {
  cb(new Error("CORS: origin not allowed"));  // everything else → 403 Forbidden
}
```

On Vercel, requests come from `https://focusarx.vercel.app` AND preview URLs like
`https://focusarx-i0dpapa0c-rangwalaaliasgar55-8394s-projects.vercel.app`.
If `APP_URL` is not set in Vercel env vars, `ALLOWED_ORIGINS` is empty → **every request is blocked**.

**Fix — replace the entire CORS block:**

```typescript
// In artifacts/api-server/src/app.ts, replace the cors() call with:
app.use(
  cors({
    origin: (origin, cb) => {
      // Always allow requests with no origin (server-to-server, curl, Postman)
      if (!origin) { cb(null, true); return; }
      
      // Allow all Vercel domains for this project
      if (
        origin.endsWith(".vercel.app") ||
        origin.endsWith(".replit.dev") ||
        origin.includes("localhost") ||
        origin.includes("127.0.0.1")
      ) {
        cb(null, true);
        return;
      }
      
      // Allow explicitly configured APP_URL
      const appUrl = process.env.APP_URL;
      if (appUrl && origin === appUrl) {
        cb(null, true);
        return;
      }
      
      // In development, allow everything
      if (process.env.NODE_ENV !== "production") {
        cb(null, true);
        return;
      }
      
      cb(new Error("CORS: origin not allowed"));
    },
    credentials: true,
  }),
);
```

### CAUSE 2 — `AUTH_SECRET` missing in Vercel → JWT signing fails → 503 → CORS error cascade

**File:** `artifacts/api-server/src/lib/config.ts`

If `AUTH_SECRET` is not set in Vercel environment variables, `jwtSecretOrRespond()` in the auth route returns `null` and sends a 503. But the CORS headers are added AFTER this response sometimes — so the browser sees a CORS error instead of the actual 503.

**Fix — add `AUTH_SECRET` to Vercel env vars NOW:**

Go to: Vercel Dashboard → Your Project → Settings → Environment Variables

Add these (all environments: Production, Preview, Development):

| Name | Value |
|------|-------|
| `AUTH_SECRET` | Run `openssl rand -base64 32` and paste the output (any 32+ char random string works) |
| `DATABASE_URL` | Your Neon/Supabase PostgreSQL URL |
| `APP_URL` | `https://focusarx.vercel.app` |
| `ADMIN_PASSWORD` | Any password you want for /admin |

**After adding env vars, redeploy** (Vercel → Deployments → click "..." → Redeploy).

---

### CAUSE 3 — Helmet CSP blocks API calls made from the browser

**File:** `artifacts/api-server/src/app.ts`

Helmet's `contentSecurityPolicy` has `connectSrc: ["'self'", "https:"]`. On Vercel, the API is on the same domain, so this should be fine. But to be safe, ensure the CSP doesn't interfere:

```typescript
// In the helmet() config, update connectSrc:
connectSrc: ["'self'", "https:", "http:"],
```

## COMPLETE FIXED `artifacts/api-server/src/app.ts`

Replace the entire file with this:

```typescript
import express, { type Express } from "express";
import cors from "cors";
import helmet from "helmet";
import pinoHttp from "pino-http";
import router from "./routes";
import { logger } from "./lib/logger";
import { getConfigErrors } from "./lib/config";
import { generalLimiter } from "./lib/rateLimiter";

const isDev = process.env.NODE_ENV !== "production";

const app: Express = express();

app.set("trust proxy", 1);

app.use(
  helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        scriptSrc: ["'self'", "'unsafe-inline'"],
        styleSrc: ["'self'", "'unsafe-inline'"],
        imgSrc: ["'self'", "data:", "https:"],
        connectSrc: ["'self'", "https:", "http:"],
        frameSrc: ["'none'"],
        objectSrc: ["'none'"],
        upgradeInsecureRequests: isDev ? null : [],
      },
    },
    crossOriginEmbedderPolicy: false,
  }),
);

app.use(
  pinoHttp({
    logger,
    serializers: {
      req(req) {
        return { id: req.id, method: req.method, url: req.url?.split("?")[0] };
      },
      res(res) {
        return { statusCode: res.statusCode };
      },
    },
  }),
);

// CORS — allow all Vercel preview URLs + configured APP_URL
app.use(
  cors({
    origin: (origin, cb) => {
      if (!origin) { cb(null, true); return; }
      if (isDev) { cb(null, true); return; }
      
      // Allow all *.vercel.app domains (covers production + all preview deployments)
      if (origin.endsWith(".vercel.app")) { cb(null, true); return; }
      
      // Allow Replit domains
      if (origin.endsWith(".replit.dev") || origin.endsWith(".repl.co")) { cb(null, true); return; }
      
      // Allow explicitly configured APP_URL
      const appUrl = process.env.APP_URL;
      if (appUrl && origin === appUrl) { cb(null, true); return; }
      
      cb(new Error("CORS: origin not allowed"));
    },
    credentials: true,
  }),
);

app.use(express.json({ limit: "100kb" }));
app.use(express.urlencoded({ extended: true, limit: "100kb" }));

app.use("/api", generalLimiter);

app.use("/api", (req, res, next) => {
  if (req.path === "/healthz") { next(); return; }
  const missing = getConfigErrors();
  if (missing.length > 0) {
    res.status(503).json({
      error: "Server is missing required configuration",
      missing,
      hint: "Add these in Vercel → Settings → Environment Variables: " + missing.join(", "),
    });
    return;
  }
  next();
});

app.use("/api", router);

app.use((err: Error, _req: express.Request, res: express.Response, _next: express.NextFunction) => {
  if (err.message?.startsWith("CORS")) {
    res.status(403).json({ error: "Forbidden", reason: err.message });
    return;
  }
  logger.error({ err }, "unhandled error");
  res.status(500).json({ error: "Internal error" });
});

export default app;
```
